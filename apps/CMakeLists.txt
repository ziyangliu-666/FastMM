function(fastmm_app name)
  add_executable(${name} ${ARGN})
  fastmm_target_defaults(${name})
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
endfunction()
